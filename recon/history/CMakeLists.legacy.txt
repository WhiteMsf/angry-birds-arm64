# Stage 24: first live Android window integration.  This is no longer a
# /data/local/tmp pbuffer executable: it is the native shared library loaded
# by android.app.NativeActivity.  The recovered Level1 Lua/Box2D/GLES1/camera/
# interpolation/KA3D draw order are rendered into a real ANativeWindow and
# Android MotionEvents are published to original updateGame() as LBUTTON/cursor.
if(ANDROID AND DEFINED BOX2D212_ROOT)
  if(DEFINED CMAKE_ANDROID_NDK)
    set(STAGE24_NATIVE_APP_GLUE "${CMAKE_ANDROID_NDK}/sources/android/native_app_glue")
  else()
    set(STAGE24_NATIVE_APP_GLUE "${ANDROID_NDK}/sources/android/native_app_glue")
  endif()
  if(NOT EXISTS "${STAGE24_NATIVE_APP_GLUE}/android_native_app_glue.c")
    message(FATAL_ERROR "android_native_app_glue.c not found under ${STAGE24_NATIVE_APP_GLUE}")
  endif()

  add_library(stage24_live_surface SHARED
    stage24_live_surface.cpp
    "${STAGE24_NATIVE_APP_GLUE}/android_native_app_glue.c"
  )
  target_include_directories(stage24_live_surface BEFORE PRIVATE
    ${LUA51_SRC}
    ${BOX2D212_ROOT}/Box2D
    "${STAGE24_NATIVE_APP_GLUE}"
  )
  set(STAGE24_DISPLAY_MODE "wvga854" CACHE STRING "Stage24 test presentation: wvga854 or native")
  set_property(CACHE STAGE24_DISPLAY_MODE PROPERTY STRINGS wvga854 native)
  option(STAGE24282_GLOBAL_SPRITE_REGISTRY "ARMv7 two-string Resources::drawSprite uses global sprite registry" OFF)
  if(STAGE24282_GLOBAL_SPRITE_REGISTRY)
    target_compile_definitions(stage24_live_surface PRIVATE STAGE24282_GLOBAL_SPRITE_REGISTRY=1)
  else()
    target_compile_definitions(stage24_live_surface PRIVATE STAGE24282_GLOBAL_SPRITE_REGISTRY=0)
  endif()
  message(STATUS "Stage24.28.2 global sprite registry contract: ${STAGE24282_GLOBAL_SPRITE_REGISTRY}")
  if(STAGE24_DISPLAY_MODE STREQUAL "native")
    target_compile_definitions(stage24_live_surface PRIVATE
      STAGE24_LIVE=1 STAGE24_DEBUG_LOGICAL_W=0 STAGE24_DEBUG_LOGICAL_H=0)
  elseif(STAGE24_DISPLAY_MODE STREQUAL "wvga854")
    target_compile_definitions(stage24_live_surface PRIVATE
      STAGE24_LIVE=1 STAGE24_DEBUG_LOGICAL_W=854 STAGE24_DEBUG_LOGICAL_H=480)
  else()
    message(FATAL_ERROR "Unknown STAGE24_DISPLAY_MODE='${STAGE24_DISPLAY_MODE}' (expected wvga854 or native)")
  endif()
  message(STATUS "Stage24 display mode: ${STAGE24_DISPLAY_MODE}")
  # Stage24.47.5: keep inline physics arithmetic in the live bridge under the
  # same non-contracted float policy as box2d212.
  target_compile_options(stage24_live_surface PRIVATE -ffp-contract=off)
  target_link_libraries(stage24_live_surface PRIVATE
    lua51_angry box2d212 android EGL GLESv1_CM OpenSLES mediandk log dl m
  )
  set_target_properties(stage24_live_surface PROPERTIES OUTPUT_NAME angryarm64)
endif()
